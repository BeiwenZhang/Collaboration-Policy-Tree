def PredictTeammateThenPlan(state):
    # Helper function to infer teammate's current skill
    def infer_teammate_skill(state):
        teammate_hold = state['hold'][0]  # teammate's holding
        any_pot_not_full = state['any_pot_not_full']
        any_soup_ready = state['any_soup_ready']

        if teammate_hold == 'soup':
            return 'deliver_soup'
        elif teammate_hold == 'dish':
            return 'fill_dish_with_soup'
        elif teammate_hold == 'onion':
            if any_pot_not_full:
                return 'put_onion_in_pot'
            else:
                return 'place_obj_on_counter'
        elif teammate_hold == 'empty':
            if any_soup_ready:
                return 'pickup_dish'
            elif any_pot_not_full:
                return 'pickup_onion'
            else:
                return None
        return None

    # Helper function to determine self action
    def determine_self_action(state, inferred_teammate_skill):
        self_hold = state['hold'][1]  # self's holding
        any_pot_not_full = state['any_pot_not_full']
        any_soup_ready = state['any_soup_ready']

        if self_hold == 'soup':
            return 'deliver_soup'
        elif self_hold == 'dish':
            if any_soup_ready:
                return 'fill_dish_with_soup'
            else:
                return 'place_obj_on_counter'
        elif self_hold == 'onion':
            if any_pot_not_full:
                return 'put_onion_in_pot'
            else:
                return 'place_obj_on_counter'
        elif self_hold == 'empty':
            if any_soup_ready:
                if inferred_teammate_skill not in ['fill_dish_with_soup', 'deliver_soup', 'pickup_dish']:
                    return 'pickup_dish'
                else:
                    return None
            elif any_pot_not_full:
                return 'pickup_onion'
            else:
                return None
        return None

    # Infer teammate's current skill
    inferred_teammate_current_skill = infer_teammate_skill(state)

    # Determine self action
    self_action_now = determine_self_action(state, inferred_teammate_current_skill)

    return inferred_teammate_current_skill, self_action_now
